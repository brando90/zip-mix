\section{Method}\label{sec:method}

\subsection{Notation}\label{sec:notation}
Let $\cD=\{d_1,\dots,d_N\}$ be the unlabeled corpus and $\cV$ a small validation suite whose tasks define the desired prior.
After the compression–ratio split (Sec.~\ref{sec:buckets}) we obtain a set of buckets
\[
  \cB \;=\;\bigl\{B_1,\dots,B_{\lvert\cB\rvert}\bigr\},
\]
where $\lvert\cB\rvert$ denotes the number of buckets.
We write $\ell_{\theta}(x)$ for the per-token negative log-likelihood of a language model with parameters $\theta$ and use $\mathrm{NCD}_{\!\mathrm{zip}}(\cdot,\cdot)$ for the normalised compression distance.

\subsection{Step 1: Compression buckets}\label{sec:buckets}
For each document $d\in\cD$ compute the compression ratio
\begin{equation}
\mathrm{CR}(d)=
\frac{\text{bytes}_{\text{zip}}(d)}
     {\text{bytes}_{\text{raw}}(d)}.
\end{equation}
A sequence of thresholds $0=c_0<c_1<\dots<c_{\lvert\cB\rvert}=1$ induces the buckets $B_m=\{d\!:\!\mathrm{CR}(d)\in[c_{m-1},c_m)\}$.

\subsection{Step 2: Alignment prior}\label{sec:prior}
Define the ZIP-FIT similarity as the maximum similarity between document $d$ and any item $v$ in the validation suite:
\[
  s(d)=\max_{v\in\cV}\bigl[1-\mathrm{NCD}_{\!\mathrm{zip}}(d,v)\bigr].
\]
The alignment weight for bucket $m$ is
\begin{equation}\label{eq:alpha0-b}
\alpha^{(0)}_m \;=\;
\frac{\sum_{d\in B_m}s(d)+\tau}
     {\sum_{B_j\in\cB}\sum_{d\in B_j}s(d)
      +\lvert\cB\rvert\,\tau},
\qquad
\tau=10^{-3},
\end{equation}
yielding $\boldsymbol{\alpha}^{(0)}\in\Delta^{\lvert\cB\rvert-1}$.

\subsection{Idea 0: Alignment-Weighted Pre-Training (AWPT)}
Training minimises
\begin{equation}\label{eq:awpt-final}
\theta^\star\;=\;
\argmin_{\theta\in\Theta}\;
\sum_{B_m\in\cB}
\alpha^{(0)}_m\;\frac{1}{\lvert B_m\rvert}
\sum_{x\in B_m} \ell_\theta(x).
\end{equation}
Stochastic optimisation uses hierarchical sampling: sample $m\!\sim\!\text{Categorical}(\boldsymbol{\alpha}^{(0)})$, then $x\!\sim\!\text{Uniform}(B_m)$.

\subsection{Idea 1: Loss-Reweighted AWPT}
Keep the dataloader uniform over $\cD$ and scale each loss term by $\alpha^{(0)}_{\,m(x)}$, where $m(x)$ maps a document to its bucket.

\subsection{Idea 2: ZipMix-DRO (One-Shot Refinement)}
Starting from $\boldsymbol{\alpha}^{(0)}$, run one Group-DRO pass on a proxy model to obtain $\boldsymbol{\alpha}^{\star}$ (Alg.~\ref{alg:zipmixdro}) and then optimise Eq.~\eqref{eq:awpt-final} with $\boldsymbol{\alpha}^{\star}$.

\begin{algorithm}
\caption{ZipMix-DRO: One-Shot Group-DRO Refinement}\label{alg:zipmixdro}
\begin{algorithmic}[1]
\Require Buckets $\cB$, alignment prior $\boldsymbol{\alpha}^{(0)}$,
         proxy model $\theta_0$, step size $\eta$, steps $T$
\Ensure Refined weights $\boldsymbol{\alpha}^{\star}$
\State $\boldsymbol{\alpha} \leftarrow \boldsymbol{\alpha}^{(0)}$
\For{$t = 1,\dots,T$}
  \State Sample batch $x \sim$ hierarchical sampler$(\boldsymbol{\alpha})$
  \State Update $\theta$ with gradient step on $\ell_\theta(x)$
  \State $\hat{\ell}_m \leftarrow$ average loss on bucket $B_m$ \Comment{excess loss}
  \State $\alpha_m \leftarrow \alpha_m \exp(\eta\,\hat{\ell}_m)$;
         normalise $\boldsymbol{\alpha}$
\EndFor
\State \Return $\boldsymbol{\alpha}^{\star} \leftarrow \boldsymbol{\alpha}$
\end{algorithmic}
\end{algorithm}

\subsection{Scaling constraint}\label{sec:scaling}
We limit the target model to at most $10^{3}$\,$\times$ the proxy size, remaining below the empirical ``dark-matter'' regime where scaling laws deviate \citep{gpt4report,schaeffer2024predictive}.
